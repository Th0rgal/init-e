theorem originalRuntimeCode_opExp_lookup :
    originalRuntimeCode.lookup (ofString "op_exp") =
      some ([],sourceFunctionBody Guest.guestFn_op_exp,.one) := by
  change (HolFiniteMapExact.empty.updateList (functionsHOL sourceDeclarations)).lookup _ = _
  calc
    _ = List.lookup (ofString "op_exp") (functionsHOL sourceDeclarations).reverse := by
      rw [HolFiniteMapExact.lookup_updateList]
      change FUPDATE_LIST FEMPTY (functionsHOL sourceDeclarations) (ofString "op_exp") = _
      simpa only [List.reverse_reverse,FLOOKUP] using
        FLOOKUP_FUPDATE_LIST_reverse_eq_lookup (functionsHOL sourceDeclarations).reverse (ofString "op_exp")
    _ = _ := by
      let declarations := InitECandidate.Proofs.FrontendStages.Declarations.originals
      have split : declarations = declarations.take 487 ++
          InitECandidate.Proofs.FrontendStages.Declarations.original487 :: declarations.drop 488 := by kernel_rfl
      have absent : ((functionsHOL (declarations.drop 488)).reverse).all
          (fun entry => !((ofString "op_exp") == entry.1)) = true := by decide_cbv
      have head : functionsHOL
          (InitECandidate.Proofs.FrontendStages.Declarations.original487 :: declarations.drop 488) =
          (ofString "op_exp",[],sourceFunctionBody Guest.guestFn_op_exp,ShapeHOL.one) ::
            functionsHOL (declarations.drop 488) := by
        rw [InitECandidate.Proofs.FrontendStages.Declarations.original487_eq]
        simp only [Guest.guestFn_op_exp,declToHOL,functionsHOL,funDeclToHOL,
          List.map_cons,List.map_nil,paramToHOL,shapeToHOL,sourceFunctionBody]
      rw [InitECandidate.Proofs.FrontendStages.Declarations.sourceDeclarations_eq]
      change List.lookup (ofString "op_exp") (functionsHOL declarations).reverse = _
      rw [split,functionsHOL_append,head,List.reverse_append,List.reverse_cons,List.append_assoc]
      have noLookup : List.lookup (ofString "op_exp") (functionsHOL (declarations.drop 488)).reverse = none := by
        apply List.lookup_eq_none_iff.mpr
        simpa only [bne] using List.all_eq_true.mp absent
      rw [List.lookup_append,noLookup]
      change List.lookup (ofString "op_exp") (_ ++ _) = _
      simp only [List.singleton_append,List.lookup_cons,beq_self_eq_true]
#print axioms originalRuntimeCode_opExp_lookup
/-- Installed declaration lookup of the direct exponentation opcode. -/
theorem exp_opcode_original_lookup :
 originalRuntimeCode.lookup expOpcodeName=some ([],expOpcodeBody,.one) :=
 originalRuntimeCode_opExp_lookup
