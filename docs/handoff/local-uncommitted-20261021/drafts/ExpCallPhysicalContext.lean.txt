/-- Actual installed dispatch-style exponentiation call. The byte-length bound
premise is the same explicit obligation as in the body endpoint. -/
theorem exp_discard_normal_physical_context
 (s post : PanSemStateFiniteExact 64 Guest.HostMemory)
 (upper : Nat) (current : InterpreterFundingFrame) (ancestors : List InterpreterFundingFrame)
 (layout : InterpreterFundingLayout s current.address.toNat upper (current::ancestors))
 (words : EvmMemoryOperandWords) (valid : EvmMemoryOperandAt s current.address words)
 (byteLengthBound : ∀ (t : PanSemStateFiniteExact 64 Guest.HostMemory) (w : BitVec 64),
  t.locals.lookup (ofString "nb")=some (.val (.word w))→w.toNat≤32)
 (fullDomain : ∀address : BitVec 64,address.toNat<33806089496→s.memaddrs address)
 (code : s.code=originalRuntimeCode)
 (run : evaluateHOLFiniteState s (.call (some (none,none)) expOpcodeName []) = (none,post)) :
 post.locals=s.locals ∧∃amount : BitVec 64,amount≤current.gas.left ∧
 InterpreterFundingLayout post current.address.toNat upper (current.charged amount::ancestors) ∧
 interpreterFundingUsed (current::ancestors) < interpreterFundingUsed (current.charged amount::ancestors) ∧
 EvmMemoryOperandAt post current.address {words with pc:=words.pc+1,stackSize:=(words.stackSize-1)-1+1} ∧
 post.globals=s.globals ∧post.memaddrs=s.memaddrs ∧post.be=s.be ∧post.code=s.code ∧
 (∀watched : BitVec 64,current.address.toNat≤watched.toNat→watched≠current.address+0→watched≠current.address+16→watched≠current.address+64→watched≠current.address+184→post.memory watched=s.memory watched) := by
 have installed : s.code.lookup expOpcodeName=some ([],expOpcodeBody,.one) := by
  rw [code];exact exp_opcode_original_lookup
 have lookup:=pan_no_args_shape_lookup s.code.lookup expOpcodeName expOpcodeBody .one installed
 obtain ⟨value,body,bodyRun,memory,locals⟩:=evaluate_discard_normal_body_memory s post expOpcodeName [] [] expOpcodeBody HolFiniteMapExact.empty .one rfl lookup run
 obtain ⟨other,otherBody,otherRun,globals,_⟩:=evaluate_discard_normal_body s post expOpcodeName [] [] expOpcodeBody HolFiniteMapExact.empty .one rfl lookup run
 have equal : otherBody=body:=(Prod.mk.inj (bodyRun.symm.trans otherRun)).2.symm
 subst otherBody
 let entry:=callEntryStateHOLFinite s HolFiniteMapExact.empty
 have entryFunding : InterpreterFundingLayout entry current.address.toNat upper (current::ancestors):=layout.after_memory rfl rfl
 have entryPhysical : EvmMemoryOperandAt entry current.address words:=valid.after_context rfl rfl rfl
 have entryCode : entry.code=originalRuntimeCode:=code
 obtain ⟨zero,amount,enough,paid,progress,physical,bodyGlobals,domain,endian,installedKept,upperFrame⟩:=exp_opcode_body_returned_physical_context entry body upper current ancestors entryFunding words entryPhysical value byteLengthBound fullDomain entryCode bodyRun
 have metadata:=evaluateHOLFiniteState_metadata s (.call (some (none,none)) expOpcodeName [])
 rw [run] at metadata
 have postDomain : post.memaddrs=body.memaddrs:=metadata.2.1.trans domain.symm
 have postGlobals : post.globals=s.globals:=globals.trans bodyGlobals
 refine ⟨locals,amount,enough,paid.after_memory memory postDomain,progress,physical.after_context globals memory postDomain,postGlobals,metadata.2.1,metadata.2.2.1,metadata.1,?_⟩
 intro watched above notPc notCounter notLeft notUsed
 rw [memory]
 exact upperFrame watched above notPc notCounter notLeft notUsed
#print axioms exp_discard_normal_physical_context
end InitECandidate.Runtime
