/-! ## Stack swap on the clock-corrected successor state -/

/-- `fixClock` only rewrites clock/termdep, so it commutes with a stack swap. -/
theorem fixClock_stack {C F : Type} (s : WordSemStateFiniteExact 64 C F)
    (r : Option (WordSemResult 64) × WordSemStateFiniteExact 64 C F)
    (xs : List (WordSemStackFrame 64)) :
    fixClock { s with stack := xs } (r.1, { r.2 with stack := xs }) =
      (fixClock s r).1, { (fixClock s r).2 with stack := xs } := by
  rcases r with ⟨res, mid⟩
  simp only [fixClock]

/-! ## Main induction

Motive (well-founded on `(s.termdep, s.clock, sizeOf p)`, matching
`wordPathWork`'s own `termination_by`):

  `∀ s xs, sValEq s.stack xs → wordPathWork p { s with stack := xs } = wordPathWork p s`

Per-constructor plan, all state reads taken from the actual definition:

* leaves (`.skip`, `.alloc`, `.storeConsts`, `.move`, `.inst`, `.assign`, `.get`,
  `.set`, `.opCurrHeap`, `.store`, `.tick`, `.mustTerminate`, `.raise`): both
  sides are `1` for everything except `.raise`/`.alloc`, whose clauses still only
  read non-stack fields; `simp only [wordPathWork, getVar_stack, ...]`.
* `.ite`: `getVar`/`getVarImm` read only `locals` (`getVar_stack`,
  `getVarImm_stack`), so the scrutinee is identical; close with the two IHs.
* `.loop`: `cutState` touches only `locals` (`cutState_stack`), so
  `cutState (names,.ln) {s with stack := xs} = some {cutSource with stack := xs}`;
  apply the IH to `body` at that cut (its stack is `sValEq`-related to
  `cutSource.stack` by `hval`), transport the run with `evaluateStackSwap body
  cutSource` (`stackSwapPost`), and apply the IH to the `wordSemSTOP` self-call
  at the swapped mid state (same program, smaller clock -> lex decrease).
* `.seq`: apply `evaluateStackSwap first s`.  For a non-exit result it yields
  `sKeyEq s.stack middle.stack`, `middle.handler = s.handler` and
  `∀ xs, sValEq s.stack xs -> ∃ st, evaluate first {s with stack := xs} =
     (res, {middle with stack := st}) ∧ sValEq middle.stack st ∧ sKeyEq xs st`.
  Hence `fixClock {s with stack := xs} (evaluate first {s with stack := xs}) =
     {fixClock s (evaluate first s) with stack := st}` (`fixClock_stack`), the
  result kinds agree (same `res`), and the IH on `second` at `st` closes; the
  first component closes by the structural IH.  For an *exit* result (`error`,
  `break`, `continue`, `result`, `exception`, `timeOut`, `finalFfi`,
  `notEnoughSpace`) the second component is not charged (`_ => 0`), so only the
  first-component IH is needed -- this is the lead's "result-indexed successor"
  point: do not demand equality of all locals on terminal outcomes.
* `.call`: the `.call` clause reads `getVars` (locals), `wordSemBadDestArgs`,
  `s.code`, `s.stackSize`, `sptDomainEmpty`/`Nodup`, `wordSemCutEnvs names
  s.locals`, `s.clock` -- all stack-independent (`getVars_stack`,
  `cutState_stack`-style) -- and builds the callee entry
  `callEnv args1 ss (pushEnv envs handler (decClock s))`.  By
  `pushEnv_stack_swap`/`callEnv_stack_swap` the swapped entry is
  `{entry with stack := fr :: xs}` with `sValEq entry.stack (fr :: xs)`, so the
  IH applies to `prog` (structural).  The run correspondence is
  `stackSwapPost`/`evaluateStackSwap` at `entry`: same `res`, and for
  `.result location returned` the *same* `location`/`returned` (so
  `location ≠ .loc l1 l2` and `returned.length ≠ values.length` agree) with
  `middleT = {middle with stack := st}`.  The post run is
  `popEnv middle` vs `popEnv {middle with stack := st}`: the popped frame is the
  one pushed by this very call in both runs (same `envs`), so both pops succeed
  or both fail and the rebuilt locals agree, hence
  `sptDomainEqUnion popped.locals envs.1 envs.2` agrees; then the IH on
  `retCode` at `setVars values returned popped` with `sValEq`-related stacks
  (via `setVars_stack`).  The `.exception` case is the same with
  `setVar reg value middle` (via `setVar_stack`) and the handler frame read from
  `middle.stack` vs its swapped copy -- `stackSwapPost` gives the `LASTN`
  handler frame with equal keys/values, so the `handlerCode` branch agrees.

/-- Sequence case, normal completion. -/
theorem wordPathWork_seq_sValEq_variant_none {C F : Type}
    (first second : WordLangProgHOL (BitVec 64))
    (s middle : WordSemStateFiniteExact 64 C F) (xs : List (WordSemStackFrame 64))
    (hval : WordSemStackEq.sValEq s.stack xs)
    (hev : evaluate first s = (none, middle))
    (ihFirst : wordPathWork first { s with stack := xs } = wordPathWork first s)
    (ihSecond : ∀ (m : WordSemStateFiniteExact 64 C F)
      (st : List (WordSemStackFrame 64)),
      WordSemStackEq.sValEq m.stack st →
      wordPathWork second { m with stack := st } = wordPathWork second m) :
    wordPathWork (.seq first second) { s with stack := xs } =
      wordPathWork (.seq first second) s := by
  have hT := Flapjack.WordSemStackEq.evaluateStackSwap first s
  rw [WordSemStackEq.stackSwapPost_iff, hev] at hT
  dsimp only [WordSemStackEq.stackSwapRel] at hT
  obtain ⟨-, -, hrun⟩ := hT
  obtain ⟨st, hevT, hvalT, -⟩ := hrun xs hval
  have hL : wordPathWork (.seq first second) { s with stack := xs } =
      1 + wordPathWork first s +
        wordPathWork second { (fixClock s ((none, middle) :
          Option (WordSemResult 64) × WordSemStateFiniteExact 64 C F)).2 with stack := st } := by
    rw [wordPathWork, hevT, ihFirst]
    dsimp only [fixClock, Prod.fst, Prod.snd]
  have hR : wordPathWork (.seq first second) s =
      1 + wordPathWork first s +
        wordPathWork second (fixClock s ((none, middle) :
          Option (WordSemResult 64) × WordSemStateFiniteExact 64 C F)).2 := by
    rw [wordPathWork]
    dsimp only [fixClock, Prod.fst, Prod.snd]
  rw [hL, hR, fixClock_stack_proj s ((none, middle) : _), ihSecond middle st hvalT]

/-- Sequence case, exiting result. -/
theorem wordPathWork_seq_sValEq_variant_result {C F : Type}
    (first second : WordLangProgHOL (BitVec 64))
    (s middle : WordSemStateFiniteExact 64 C F) (xs : List (WordSemStackFrame 64))
    (location : WordLocW 64) (returned : List (WordLocW 64))
    (hval : WordSemStackEq.sValEq s.stack xs)
    (hev : evaluate first s = (some (.result location returned), middle))
    (ihFirst : wordPathWork first { s with stack := xs } = wordPathWork first s) :
    wordPathWork (.seq first second) { s with stack := xs } =
      wordPathWork (.seq first second) s := by
  have hT := Flapjack.WordSemStackEq.evaluateStackSwap first s
  rw [WordSemStackEq.stackSwapPost_iff, hev] at hT
  dsimp only [WordSemStackEq.stackSwapRel] at hT
  obtain ⟨-, -, hrun⟩ := hT
  obtain ⟨st, hevT, -, -⟩ := hrun xs hval
  have hL : wordPathWork (.seq first second) { s with stack := xs } =
      1 + wordPathWork first s := by
    rw [wordPathWork, hevT, ihFirst]
    dsimp only [fixClock, Prod.fst, Prod.snd]
  have hR : wordPathWork (.seq first second) s = 1 + wordPathWork first s := by
    rw [wordPathWork]
    dsimp only [fixClock, Prod.fst, Prod.snd]
  rw [hL, hR]
